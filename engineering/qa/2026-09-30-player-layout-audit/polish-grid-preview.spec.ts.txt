import { expect, test } from "@playwright/test";
import { configureLayoutAudit, login } from "./layout-audit-helpers";
configureLayoutAudit();
test("measure WebKit settings label grid", async ({ page }, testInfo) => {
 await login(page);
 await page.setViewportSize({ width: 320, height: 800 });
 await page.goto("/settings#transcoder", { waitUntil: "domcontentloaded" });
 const measure = () => page.locator('select[name="codec"]').evaluate(element => ({ documentWidth: document.documentElement.scrollWidth, viewportWidth: innerWidth, labelTrack: getComputedStyle(element.parentElement!).gridTemplateColumns, labelWidth: element.parentElement!.getBoundingClientRect().width, selectWidth: element.getBoundingClientRect().width, selectMinimum: getComputedStyle(element).minWidth }));
 const before = await measure();
 await page.screenshot({ path: testInfo.outputPath("320-before.png") });
 const index = await page.evaluate(() => {
  const sheet = [...document.styleSheets].find(sheet => sheet.href?.includes("/static/app.css"))!;
  return sheet.insertRule(".settings-shell .settings-flow label{grid-template-columns:minmax(0,1fr)}", sheet.cssRules.length);
 });
 const after = await measure();
 expect(after.documentWidth).toBe(after.viewportWidth);
 await page.screenshot({ path: testInfo.outputPath("320-preview.png") });
 await page.evaluate(index => [...document.styleSheets].find(sheet => sheet.href?.includes("/static/app.css"))!.deleteRule(index), index);
 const restored = await measure();
 const geometry = { before, after, restored };
 console.log(JSON.stringify(geometry));
 await testInfo.attach("grid-geometry", { body: JSON.stringify(geometry, null, 2), contentType: "application/json" });
});
