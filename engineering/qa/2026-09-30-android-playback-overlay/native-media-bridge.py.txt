from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
import http.client
import json
import ssl
import threading

ROOT = Path(__file__).resolve().parents[1]
TLS = ssl.create_default_context(cafile=str(ROOT / "host/data/tls-ca.pem"))
MEDIA_READY = threading.Event()
MEDIA_READY.set()
MEDIA_MODE = "ready"
LOCK = threading.Lock()


class Bridge(BaseHTTPRequestHandler):
    def do_GET(self):
        self.forward()

    def do_PUT(self):
        self.forward()

    def do_POST(self):
        global MEDIA_MODE
        if self.path != "/__fixture/media" or self.headers.get("X-Fixture-Control") != "video-overlay":
            self.send_error(400)
            return
        try:
            size = int(self.headers.get("Content-Length", "0"))
            if not 0 < size <= 128:
                raise ValueError()
            value = json.loads(self.rfile.read(size))
            if set(value) != {"mode"} or value["mode"] not in {"ready", "pending", "failed"}:
                raise ValueError()
        except (ValueError, TypeError, KeyError):
            self.send_error(400)
            return
        with LOCK:
            MEDIA_MODE = value["mode"]
            if MEDIA_MODE == "pending":
                MEDIA_READY.clear()
            else:
                MEDIA_READY.set()
        self.send_response(204)
        self.end_headers()

    def forward(self):
        if len(self.path) > 4096 or not self.path.startswith(("/api/v1/", "/art/", "/backdrop/", "/episode-art/", "/media/", "/subtitle/")):
            self.send_error(400)
            return
        if self.path.startswith("/media/"):
            MEDIA_READY.wait(30)
            with LOCK:
                mode = MEDIA_MODE
            if mode != "ready":
                self.send_error(404 if mode == "failed" else 503)
                return
        try:
            size = int(self.headers.get("Content-Length", "0"))
            if not 0 <= size <= 65536:
                raise ValueError()
        except ValueError:
            self.send_error(400)
            return
        upstream = http.client.HTTPSConnection("127.0.0.1", 39227, context=TLS, timeout=30)
        try:
            body = self.rfile.read(size) if size else None
            headers = {k: v for k, v in self.headers.items() if k.lower() in ("authorization", "accept", "range", "if-range", "content-type", "x-kinosail-viewer-profile")}
            upstream.request(self.command, self.path, body=body, headers=headers)
            response = upstream.getresponse()
            data = response.read(8388609)
            if len(data) > 8388608:
                self.send_error(502)
                return
            self.send_response(response.status)
            for name in ("Content-Type", "Content-Range", "Accept-Ranges"):
                value = response.getheader(name)
                if value:
                    self.send_header(name, value)
            self.send_header("Content-Length", str(len(data)))
            self.end_headers()
            self.wfile.write(data)
        except (OSError, http.client.HTTPException):
            self.send_error(502)
        finally:
            upstream.close()

    def log_message(self, *args):
        pass


ThreadingHTTPServer(("127.0.0.1", 39231), Bridge).serve_forever()
