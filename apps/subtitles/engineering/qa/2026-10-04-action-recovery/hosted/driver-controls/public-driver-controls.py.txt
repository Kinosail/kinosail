#!/usr/bin/env python3
"""Prepared public driver E2E controls. SOURCE ONLY; do not invoke real Go."""
import argparse
import ast
import hashlib
import json
import os
from pathlib import Path
import shutil
import signal
import subprocess
import sys
import time

# These controls use a fresh small Git fixture containing only selected source
# text and the driver recipe. Its fake go executable supplies a declared event
# stream. A passing fixture stream is driver evidence, never R06 product GREEN.
# Preserve every fixture/output; no automatic directory removal is performed.

def prepare(source, directory):
    recipe = "apps/subtitles/engineering/qa/2026-10-04-action-recovery/hosted"
    helper = source / "apps/subtitles/scripts/campaign_r06_sources.py"
    assignments = {target.id: node.value.value for node in ast.parse(helper.read_text()).body
                   if isinstance(node, ast.Assign) and isinstance(node.value, ast.Constant)
                   for target in node.targets if isinstance(target, ast.Name)}
    scope_name = assignments.get("SCOPE_FILE", "protocol-scope.json")
    assert isinstance(scope_name, str) and Path(scope_name).name == scope_name
    scope = json.loads((source / recipe / scope_name).read_text())
    names = [s["path"] for s in scope["sourcePins"]]
    names += [recipe + "/" + scope_name, scope["overlay"]["baselineFixture"],
              "apps/subtitles/scripts/campaign-r06-public.py"]
    for helper in ("campaign_r06_sources.py", "campaign_r06_projection.py"):
        if (source / "apps/subtitles/scripts" / helper).exists():
            names.append("apps/subtitles/scripts/" + helper)
    for name in names:
        target = directory / name
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(source / name, target)
    baseline = directory / scope["overlay"]["baselineFixture"]
    target = directory / scope["overlay"]["replace"]
    target.write_bytes(baseline.read_bytes())
    (directory / "driver-control-marker.txt").write_text("unchanged fictional marker")
    tools = directory / "control-tools"
    tools.mkdir()
    shutil.copyfile(source / recipe / "driver-controls/fictional-go.py.txt", tools / "go")
    (tools / "go").chmod(0o700)
    for command in (["git", "init", "-q"], ["git", "add", "."],
                    ["git", "-c", "user.name=Fictional Control", "-c", "user.email=control@example.invalid", "commit", "-qm", "fictional driver inputs"]):
        subprocess.run(command, cwd=directory, check=True, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    return scope, subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=directory, text=True).strip(), scope_name

def control(source, output, name):
    root = output / name
    root.mkdir()
    scope, revision, scope_name = prepare(source, root)
    env = dict(os.environ, PATH=str(root / "control-tools") + os.pathsep + os.environ["PATH"],
               GITHUB_SHA=revision, R06_DRIVER_CASE=name, R06_DRIVER_SCOPE=str(root / "apps/subtitles/engineering/qa/2026-10-04-action-recovery/hosted" / scope_name),
               R06_DRIVER_STARTED=str(root / "fake-go-started"), R06_DRIVER_DESCENDANT=str(root / "fake-descendant"),
               R06_DRIVER_COMPILE_STARTED=str(root / "fake-compile-started"), R06_DRIVER_BINARY_STARTED=str(root / "fake-binary-started"),
               R06_DRIVER_DRIFT=str(root / "driver-control-marker.txt"))
    if name == "hosted-missing-revision":
        env.pop("GITHUB_SHA")
    elif name == "hosted-wrong-revision":
        env["GITHUB_SHA"] = "0" * 40
    command = [sys.executable, "apps/subtitles/scripts/campaign-r06-public.py"]
    if name == "default-output":
        command.append("--static-check")
        evidence = root / ".verification/campaign-proof/R06"
    else:
        evidence = root / "safe-results"
        command += ["--output", str(evidence)]
    if name == "permission-denied":
        command = [sys.executable, "-c", "import os,runpy,sys; from pathlib import Path; sys.path.insert(0,'apps/subtitles/scripts'); real=os.killpg; os.killpg=lambda *a: (_ for _ in ()).throw(PermissionError('fictional control')) if Path(os.environ['R06_DRIVER_STARTED']).exists() else real(*a); sys.argv=['apps/subtitles/scripts/campaign-r06-public.py','--output',sys.argv[1]]; runpy.run_path(sys.argv[0],run_name='__main__')", str(evidence)]
    started = time.monotonic()
    group_absent_before_cleanup = None
    try:
        result = subprocess.run(command, cwd=root, env=env, capture_output=True, timeout=240)
        if (root / "fake-go-started").exists():
            try:
                os.killpg(int((root / "fake-go-started").read_text()), 0)
                group_absent_before_cleanup = False
            except ProcessLookupError:
                group_absent_before_cleanup = True
    finally:
        # Only a PID recorded by this control's fake executable is eligible.
        for pid_file in ("fake-go-started", "fake-compile-started"):
            if not (root / pid_file).exists():
                continue
            try:
                os.killpg(int((root / pid_file).read_text()), signal.SIGKILL)
            except ProcessLookupError:
                pass
    receipt = json.loads((evidence / "receipt.json").read_text())
    results = json.loads((evidence / "results.json").read_text())
    exported = b"".join((evidence / n).read_bytes() for n in scope["exportAllowlist"])
    assert b"FAKE_TOKEN_SENTINEL" not in exported + result.stdout + result.stderr
    if name == "default-output":
        assert result.returncode == 0 and receipt["classification"] == "static-source-only"
        assert not (root / "fake-go-started").exists()
    elif name.startswith("hosted-"):
        assert result.returncode != 0 and not receipt["goWasLaunched"]
        assert not (root / "fake-go-started").exists()
    elif name == "private-sentinel":
        assert result.returncode == 0 and receipt["classification"] == "focused-public-protocol-green"
        assert receipt["ownedProcessExited"] and receipt["ownedGroupSettled"]
        assert group_absent_before_cleanup
        assert receipt["compilation"]["exitCode"] == 0 and receipt["compilation"]["ownedGroupSettled"]
        assert receipt["compiledTestBinarySHA256"] == hashlib.sha256((evidence/"work/public-server.test").read_bytes()).hexdigest()
        assert receipt["compiledBinaryUnchanged"] and receipt["publicRuntimeWasLaunched"]
        assert receipt["bounds"]["compileSeconds"] == 120 and receipt["bounds"]["goSeconds"] == 80 and receipt["bounds"]["externalSeconds"] == 90
    elif name == "leader-exit-held-stdout":
        assert result.returncode != 0 and receipt["stopReason"] == "external-timeout"
        assert 89 <= time.monotonic() - started <= 102
        assert receipt["ownedProcessExited"]
        assert receipt["ownedGroupSettled"]
        assert group_absent_before_cleanup
    elif name == "permission-denied":
        assert result.returncode != 0 and receipt["classification"] == "incomplete-or-failed"
        assert receipt["exitCode"] == 0 and receipt["ownedProcessExited"]
        assert not receipt["ownedGroupSettled"] and receipt["settlementFailureClass"] == "PermissionError"
        assert receipt["stopReason"] == "process-settlement-failure"
        assert receipt["sourcesUnchanged"] and receipt["trackedTreeUnchanged"] and receipt["R16Preserved"]
    elif name == "tracked-drift":
        assert result.returncode != 0 and not receipt["trackedTreeUnchanged"]
    elif name.startswith("compile-"):
        assert result.returncode != 0 and receipt["classification"] == "incomplete-or-failed"
        assert receipt["compilation"]["goWasLaunched"]
        if name == "compile-stream-is-not-tests":
            assert receipt["publicRuntimeWasLaunched"] and results["counts"]["topLevel"]["pass"] == 0
        else:
            assert not receipt["publicRuntimeWasLaunched"]
        if name == "compile-leader-held-stdout":
            assert receipt["compilation"]["stopReason"] == "external-timeout"
            assert 119 <= time.monotonic()-started <= 132
            assert receipt["compilation"]["ownedProcessExited"] and receipt["compilation"]["ownedGroupSettled"]
        else:
            assert results["counts"]["topLevel"]["pass"] == 0 and results["counts"]["subtests"]["pass"] == 0
    elif name == "binary-mutated-during-runtime":
        assert result.returncode != 0 and not receipt["compiledBinaryUnchanged"]
        assert results["counts"]["topLevel"]["pass"] == 33 and results["counts"]["subtests"]["pass"] == 45
    else:
        assert result.returncode != 0 and receipt["classification"] == "incomplete-or-failed"
        if name.startswith("package-"):
            assert results["duplicatePackageTerminalCount"] > 0
        else:
            assert results["invalidEventCount"] > 0
    return {"name": name, "boundary": "fictional driver control only; no actual Go or product proof", "status": "pass",
            "seconds": round(time.monotonic() - started, 3), "receiptSHA256": hashlib.sha256((evidence / "receipt.json").read_bytes()).hexdigest()}

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--source", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    args.output.mkdir(exist_ok=False)
    cases = ["default-output", "hosted-missing-revision", "hosted-wrong-revision", "private-sentinel",
             "package-fail-pass", "package-duplicate", "invalid-action", "missing-action", "boolean-test",
             "boolean-elapsed", "missing-elapsed", "tracked-drift", "permission-denied", "leader-exit-held-stdout",
             "compile-failure", "compile-missing-binary", "compile-symlink-binary", "compile-source-drift",
             "compile-stream-is-not-tests", "binary-mutated-during-runtime", "compile-leader-held-stdout"]
    records = [control(args.source.resolve(), args.output, name) for name in cases]
    (args.output / "control-receipt.json").write_text(json.dumps({"boundary": "fictional-only; source prepared before driver correction", "cases": records}, indent=2) + "\n")

if __name__ == "__main__":
    main()
